# 什么时候需要“&”? 参数结果需要“带回来”的时候

```C
#include<stdio.h>

void test1(int x)
{
	x=1024;
	printf("test1函数内部 x=%d\n",x); 
}

void test2(int &x)
{
	x=1024;
	printf("test2函数内部 x=%d\n",x); 
}



int main()
{
	int x=1;
	printf("%d\n",x);
	test1(x);
	printf("%d\n",x);
	test2(x);
	printf("%d",x);
}
```
